\section*{3. Kết quả đã đo được}
\begin{itemize}
\item Vòng tròn Cấp 1--4 (192 ván): điểm toàn giải Cấp 3 $=0{,}958$; Cấp 2 $=0{,}708$; Cấp 4 $=0{,}297$; Cấp 1 $=0{,}036$.
\item Ablation Alpha-Beta ($d=3$): cắt tỉa giảm khoảng $7{,}1\times$ số nút; sắp xếp nước đi giảm thêm khoảng $3{,}6\times$; TT thêm khoảng 2,7\%; quiescence nâng số thế WAC giải đúng từ 27 lên 37/50.
\item UCT với 25--200 mô phỏng thua toàn bộ 96 ván trước Cấp 2.
\end{itemize}
Chi tiết, bảng và biểu đồ xem báo cáo đầy đủ \code{Nhom27\_GameAI\_CoVua\_TTNTNC.pdf}.

\section*{4. Khó khăn và rủi ro}
\begin{itemize}
\item Phần học sâu (Cấp 5, 7, 8) và RL (Cấp 6) chiếm phần lớn rubric còn thiếu (mục 4 của rubric: 15 điểm cho tác tử học được); rủi ro cao nếu không kịp huấn luyện thật.
\item Số ván mỗi cặp mới là 32, thấp hơn 100 ván khuyến nghị; chưa chạy chế độ time-control.
\item Cấp 4 yếu bất thường; nguyên nhân mới ở mức giả thuyết, cần khảo sát thêm (số mô phỏng lớn hơn, rollout tốt hơn).
\item Chưa gán họ tên cho vai trò TV1--TV3 trong tài liệu phân công; chưa thống kê số commit từng người.
\end{itemize}
